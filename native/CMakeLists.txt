cmake_minimum_required(VERSION 3.24)
project(EDMStudioNative VERSION 0.12.0 LANGUAGES C CXX)
enable_language(RC)
configure_file(src/version.rc.in ${CMAKE_CURRENT_BINARY_DIR}/version.rc @ONLY)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE _CRT_SECURE_NO_WARNINGS)
add_compile_options($<$<CXX_COMPILER_ID:MSVC>:/utf-8> $<$<CXX_COMPILER_ID:MSVC>:/MP>)
set(V ${CMAKE_CURRENT_SOURCE_DIR}/vendor)
if(NOT EXISTS ${V}/imgui/imgui.cpp)
  message(FATAL_ERROR "Run native/fetch-deps.ps1 first")
endif()
add_library(imgui STATIC ${V}/imgui/imgui.cpp ${V}/imgui/imgui_draw.cpp ${V}/imgui/imgui_tables.cpp ${V}/imgui/imgui_widgets.cpp
  ${V}/imgui/backends/imgui_impl_win32.cpp ${V}/imgui/backends/imgui_impl_dx11.cpp ${V}/imgui/misc/cpp/imgui_stdlib.cpp)
target_include_directories(imgui PUBLIC ${V}/imgui ${V}/imgui/backends ${V}/imgui/misc/cpp)
file(GLOB LUA_SOURCES ${V}/lua/src/*.c)
list(FILTER LUA_SOURCES EXCLUDE REGEX "/lua(c)?\\.c$")
# Lua's C++ mode uses exceptions instead of longjmp across our C++ callbacks.
# This is required for correct unwinding of strings, paths and ZIP buffers.
set_source_files_properties(${LUA_SOURCES} PROPERTIES LANGUAGE CXX)
add_library(lua STATIC ${LUA_SOURCES})
target_include_directories(lua PUBLIC ${V}/lua/src)
if(NOT EXISTS ${CMAKE_CURRENT_BINARY_DIR}/miniz_export.h)
  file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/miniz_export.h "#define MINIZ_EXPORT\n")
endif()
add_library(miniz STATIC ${V}/miniz/miniz.c ${V}/miniz/miniz_tdef.c ${V}/miniz/miniz_tinfl.c ${V}/miniz/miniz_zip.c)
target_include_directories(miniz PUBLIC ${V}/miniz ${CMAKE_CURRENT_BINARY_DIR})
file(GLOB TEX_SOURCES ${V}/directxtex/DirectXTex/*.cpp)
list(FILTER TEX_SOURCES EXCLUDE REGEX "(D3D11|D3D12|GPU|BCDirectCompute)\\.cpp$")
add_library(dxtex STATIC ${TEX_SOURCES})
target_include_directories(dxtex PUBLIC ${V}/directxtex/DirectXTex)
target_compile_definitions(dxtex PRIVATE _WIN32_WINNT=0x0A00)
target_link_libraries(dxtex PUBLIC windowscodecs ole32)
add_library(edm_core STATIC src/common.cpp src/parser.cpp src/scene.cpp src/assets.cpp src/lua.cpp src/export.cpp src/animation_analysis.cpp)
target_sources(edm_core PRIVATE src/attachment.cpp)
target_sources(edm_core PRIVATE src/paint_assembly.cpp)
target_sources(edm_core PRIVATE src/export_obj.cpp src/export_fbx.cpp)
target_sources(edm_core PRIVATE src/paint.cpp src/paint_document.cpp)
target_sources(edm_core PRIVATE src/paint_mapping_common.cpp src/paint_wrap.cpp)
target_sources(edm_core PRIVATE src/paint_projection.cpp)
target_sources(edm_core PRIVATE src/paint_batch.cpp src/paint_auto_mapping.cpp)
target_sources(edm_core PRIVATE src/paint_decal.cpp src/paint_conform.cpp)
target_sources(edm_core PRIVATE src/paint_quality.cpp)
target_include_directories(edm_core PUBLIC src ${V}/eigen ${V}/json/single_include)
target_link_libraries(edm_core PUBLIC lua miniz dxtex shell32 ole32 advapi32)
target_compile_definitions(edm_core PUBLIC EIGEN_MPL2_ONLY EDM_NATIVE_VERSION="${PROJECT_VERSION}")
add_executable(EDM-Studio-Native WIN32 src/app.cpp src/renderer.cpp src/paint_editor.cpp ${CMAKE_CURRENT_BINARY_DIR}/version.rc)
target_link_libraries(EDM-Studio-Native PRIVATE edm_core imgui d3d11 dxgi dxguid d3dcompiler dwmapi comdlg32)
add_executable(edm-native-cli src/cli.cpp ${CMAKE_CURRENT_BINARY_DIR}/version.rc)
target_link_libraries(edm-native-cli PRIVATE edm_core)
add_executable(edm-native-tests tests/tests.cpp tests/regression.cpp)
target_link_libraries(edm-native-tests PRIVATE edm_core)
target_link_options(edm-native-tests PRIVATE /MAP:${CMAKE_CURRENT_BINARY_DIR}/edm-native-tests.map)
target_compile_definitions(edm-native-tests PRIVATE EDM_TEST_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
enable_testing()
add_test(NAME native_core COMMAND edm-native-tests)
